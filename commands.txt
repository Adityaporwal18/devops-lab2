Q1 - CONTAINERIZE A SAMPLE APPLICATION USING DOCKER

[E01] Docker apt repository
  ~/devops-lab1$ cd ~
  ~$ sudo install -m 0755 -d /etc/apt/keyrings
  ~$ sudo curl -fsSL https://download.docker.com/linux/ubuntu/gpg -o /etc/apt/keyrings/docker.asc
  ~$ sudo chmod a+r /etc/apt/keyrings/docker.asc
  ~$ echo "deb [arch=$(dpkg --print-architecture) signed-by=/etc/apt/keyrings/docker.asc] https://download.docker.com/linux/ubuntu $(. /etc/os-release && echo "$VERSION_CODENAME") stable" | sudo tee /etc/apt/sources.list.d/docker.list > /dev/null
  ~$ sudo apt update

Q3 - DOCKER STORAGE
-------------------

[E15] Container-local data is lost
  ~/devops-lab2$ docker run --name storage-test ubuntu sh -c 'echo "Temporary Lab 2 Data" > /tmp/data.txt && cat /tmp/data.txt'
  ~/devops-lab2$ docker rm storage-test
  ~/devops-lab2$ docker run --rm ubuntu cat /tmp/data.txt

[E16] Named volume keeps data
  ~/devops-lab2$ docker volume create lab2-data
  ~/devops-lab2$ docker run --name volume-test -v lab2-data:/data ubuntu sh -c 'echo "Persistent Lab 2 Data" > /data/data.txt'
  ~/devops-lab2$ docker rm volume-test
  ~/devops-lab2$ docker run --rm -v lab2-data:/data ubuntu cat /data/data.txt
  ~/devops-lab2$ docker volume ls

[E17] Bind mount shares files both ways
  ~/devops-lab2$ mkdir -p ~/devops-lab2/bind-data
  ~/devops-lab2$ echo "Created from Ubuntu host" > ~/devops-lab2/bind-data/host.txt
  ~/devops-lab2$ docker run --rm -v ~/devops-lab2/bind-data:/data ubuntu cat /data/host.txt
  ~/devops-lab2$ docker run --rm -v ~/devops-lab2/bind-data:/data ubuntu sh -c 'echo "Created from Docker container" > /data/container.txt'
  ~/devops-lab2$ cat ~/devops-lab2/bind-data/container.txt
  ~/devops-lab2$ ls -l ~/devops-lab2/bind-data

Q4 - DOCKER NETWORK
-------------------

[E18] Ping by container name
  ~/devops-lab2$ docker network create lab2-network
  ~/devops-lab2$ docker run -d --name network-a --network lab2-network alpine sleep 3600
  ~/devops-lab2$ docker run -d --name network-b --network lab2-network alpine sleep 3600

Q5 - DEPLOY AN APPLICATION TO KUBERNETES
----------------------------------------

[E22] Install kubectl and Minikube
  ~/devops-lab2$ cd ~
  ~$ curl -sLO "https://dl.k8s.io/release/$(curl -L -s https://dl.k8s.io/release/stable.txt)/bin/linux/arm64/kubectl"
  ~$ sudo install -o root -g root -m 0755 kubectl /usr/local/bin/kubectl
  ~$ curl -sLO https://storage.googleapis.com/minikube/releases/latest/minikube-linux-arm64
  ~$ sudo install minikube-linux-arm64 /usr/local/bin/minikube
  ~$ rm kubectl minikube-linux-arm64
  ~$ kubectl version --client
  ~$ minikube version

[E23] Start Minikube

  ~$ kubectl get nodes
  ~$ minikube status

  ~$ kubectl get pods -n kube-system

[E25] Apply the Deployment and Service
